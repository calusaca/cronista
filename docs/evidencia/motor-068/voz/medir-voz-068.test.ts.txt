// TEMPORAL (brief 068 § 3, medición en reinicios-v10; no se commitea como prueba): consumo de las bolsas de voz de un directo con narración a 60, 120 y
// 180 s, con Speech simulado y el narrador fiel (stub), recorriendo como el consumidor: previa, apertura, tramos del
// narrador, intervenciones de la mesa y cierre, por idioma. Suma las reservas de extras.json por clase.
import { readFileSync } from "node:fs";
import { mkdtemp, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, describe, it, vi } from "vitest";
import { planProgramado } from "../controladores";
import type { ModeloConfigurado } from "../modelos";
import { GestorSesiones, PRESUPUESTO_AUDIO_ENCUENTRO } from "../sesiones";
import { POST as relatoHttp } from "../../app/api/arena/sesiones/[id]/relato/route";
import type { HechoVerificable } from "../../cadena/verificador";
import type { Llm } from "../../lib/llm";
import { MESA_ARENA, PREVIA_ARENA } from "../../sim/limites";
import { previaPartido } from "../previa";

describe("068 · medición de las bolsas de voz en v10 (temporal)", () => {
  const rutas: string[] = [];
  const compartido = globalThis as typeof globalThis & { __cronistaSesiones?: GestorSesiones };
  let previo: GestorSesiones | undefined;
  afterEach(async () => { compartido.__cronistaSesiones = previo; vi.unstubAllEnvs(); vi.unstubAllGlobals(); await Promise.all(rutas.splice(0).map(p => rm(p, { recursive: true, force: true }))); });
  const llmDe = (completar: Llm["completar"]): Llm => ({ id: "modelo-prueba", completar });
  const fielLlm = () => llmDe(async (_s, m) => { const c = JSON.parse(m[0]!.contenido) as { hechos: HechoVerificable[]; principal: string }, h = c.hechos.find(h => h.id === c.principal)!; return JSON.stringify({ oraciones: [{ texto_es: h.texto_es, texto_en: h.texto_en, hecho: h.id, apoyo: [] }] }); });
  const pedir = (id: string, cuerpo: object) => relatoHttp(new Request("http://localhost/relato", { method: "POST", body: JSON.stringify(cuerpo) }), { params: Promise.resolve({ id }) });

  const medir = async (seed: number, segundos: number, idiomas: readonly ("es" | "en")[], variantes: readonly ("breve" | "completa")[]) => {
    previo = compartido.__cronistaSesiones;
    vi.stubEnv("ARENA_BLOB_ENDPOINT", ""); vi.stubEnv("ARENA_VOZ", "1");
    vi.stubEnv("AZURE_SPEECH_ENDPOINT", "https://ejemplo.cognitiveservices.azure.com"); vi.stubEnv("AZURE_SPEECH_KEY", "prueba");
    vi.stubEnv("ARENA_SPEECH_DIARIO_CARACTERES", "100000000"); vi.stubEnv("ARENA_SPEECH_DIARIO_AUDIOS", "1000000");
    const directorio = await mkdtemp(join(tmpdir(), "cronista-voz-068-")); rutas.push(directorio);
    const catalogo: ModeloConfigurado[] = [{ publico: { id: "p", nombre: "P", controlador: "p" }, crear: () => llmDe(async (_s, m) => JSON.stringify(planProgramado(JSON.parse(m[0]!.contenido)))) }];
    const g = new GestorSesiones({ directorio, ritmoMsPorSegundo: 0, catalogo: () => catalogo, narrador: () => fielLlm() }); compartido.__cronistaSesiones = g;
    const { id } = await g.crear({ seed, segundos, local: "auto", visita: "auto", seleccion: { modo: "manual", local: "p", visita: "p" }, presentacion: "buffer" });
    await g.esperarFinal(id);
    vi.stubGlobal("fetch", vi.fn<typeof fetch>(async url => String(url).includes("/voices/list") ? new Response("x", { status: 500 }) : new Response(new Uint8Array([1, 2, 3]))));
    const r = await g.reproduccion(id), fin = r.registro.final.tick;
    type Paso = { tick: number; orden: number; tipo: string; cuerpo: object };
    const plan: Paso[] = [];
    for (const idioma of idiomas) {
      for (const i of previaPartido(r.registro)) for (const variante of variantes) plan.push({ tick: 0, orden: 0, tipo: "previa", cuerpo: { desdeTick: 0, hastaTick: 0, modo: "buffer", idioma, intervencion: i.id, variante } });
      plan.push({ tick: 0, orden: 0.5, tipo: "narrador", cuerpo: { desdeTick: 0, hastaTick: Math.min(50, fin), modo: "buffer", idioma, entradaTick: 0 } });
      for (let desde = 0; desde < fin; desde += 50) plan.push({ tick: desde, orden: 1, tipo: "narrador", cuerpo: { desdeTick: desde, hastaTick: Math.min(desde + 50, fin), modo: "buffer", idioma } });
      for (const i of r.intervenciones) for (const variante of variantes) if (variante === "completa" || i.turnosBreve)
        plan.push({ tick: i.tick, orden: 2, tipo: "invitado", cuerpo: { desdeTick: i.tick, hastaTick: i.tick, modo: "buffer", idioma, intervencion: i.id, variante } });
      plan.push({ tick: fin + 1, orden: 3, tipo: "cierre", cuerpo: { desdeTick: Math.floor((fin - 1) / 50) * 50, hastaTick: fin, modo: "buffer", idioma, cierre: true } });
    }
    plan.sort((a, b) => a.tick - b.tick || a.orden - b.orden);
    const pedidos: Record<string, { pedidas: number; ok: number; rechazos: Record<string, number> }> = {};
    for (const p of plan) {
      const res = await pedir(id, p.cuerpo), x = pedidos[p.tipo] ??= { pedidas: 0, ok: 0, rechazos: {} };
      x.pedidas++;
      if (res.status === 200) x.ok++; else { const e = `${res.status} ${(await res.json()).error}`; x.rechazos[e] = (x.rechazos[e] ?? 0) + 1; }
    }
    const reservas = Object.values(JSON.parse(readFileSync(join(directorio, `${id}.extras.json`), "utf8")) as Record<string, { tipo: string; clase?: string; caracteres?: number }>).filter(x => x.tipo === "audio");
    const por = (f: (c: string) => boolean) => { const xs = reservas.filter(x => f(x.clase ?? "narrador")); return { solicitudes: xs.length, caracteres: xs.reduce((s, x) => s + (x.caracteres ?? 0), 0) }; };
    return { seed, segundos, idiomas: idiomas.join("+"), variantes: variantes.join("+"), intervenciones: r.intervenciones.length,
      narradorYCierre: por(c => c === "narrador" || c === "cierre"), invitados: por(c => c === "invitado"), previa: por(c => c === "previa"), pedidos };
  };

  it("mide 120 s × ES y ES+EN × una variante (completa), semillas 1–8, con ARENA.reglas (reinicios-v10)", async () => {
    console.log("[068 voz · bolsas]", JSON.stringify({ narrador: PRESUPUESTO_AUDIO_ENCUENTRO, reservaCierre: { solicitudes: MESA_ARENA.dialogo.reservaSpeechCierreSolicitudes, caracteres: MESA_ARENA.dialogo.reservaSpeechCierreCaracteres }, invitados: MESA_ARENA.dialogo.bolsaSpeechInvitados, previa: PREVIA_ARENA.bolsaSpeech }));
    for (const segundos of [120]) for (const idiomas of [["es"], ["es", "en"]] as const) for (const variantes of [["completa"]] as const) for (let seed = 1; seed <= 8; seed++)
      console.log("[068 voz]", JSON.stringify(await medir(seed, segundos, idiomas, variantes)));
  }, 3_600_000);
});
